   - _resolve_value(self, doc: dict[str, Any]) -> Any  # 支持点号分隔的嵌套字段访问（如 fire_wall.fire_resistance_rating）。

   - _default_rules_dir() -> Path  # 返回包内 rules/ 目录路径。

   - _load_one(self, path: Path) -> list[StructuredRule]
   - check(self, design_doc: dict[str, Any]) -> list[Violation]  # 对设计文档执行全部规则检查，返回违规列表。
   - rules_count(self) -> int
